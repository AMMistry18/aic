project(research_grasp LANGUAGES CXX)
find_package(gz-sim9 REQUIRED)
find_package(gz-plugin3 REQUIRED COMPONENTS register)
add_library(ResearchGrasp SHARED ResearchGrasp.cc)
target_compile_features(ResearchGrasp PRIVATE cxx_std_17)

target_compile_features(ResearchContactAudit PRIVATE cxx_std_17)
